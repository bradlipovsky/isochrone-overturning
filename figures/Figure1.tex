% The two panels use the incompressible coordinate map
%   x = x_0 + tau U(x_0),  z = s(x_0) - 0.3 tau U(x_0),
% where s(x_0)=5-0.15x_0 and
% U(x_0)=3-0.7 tanh[(x_0-3)/0.7].
% Its surface accumulation is 0.15U>0, its horizontal speed is positive,
% and its coordinate determinant is 0.15U>0. The order Jacobian is
% J=1+tau U'(x_0): it is positive at tau=0.8 and locally negative at tau=2.5.

\pgfmathdeclarefunction{speed}{1}{%
  \pgfmathparse{3 - 0.7*tanh((#1-3)/0.7)}%
}

\begin{tikzpicture}[
  x=0.52cm,
  y=0.76cm,
  >=Latex,
  font=\small,
  labelbox/.style={fill=white,inner sep=1.2pt},
  trajectory/.style={->,thick},
  isochrone/.style={dashed,very thick}
]

  \pgfmathsetmacro{\sourcea}{2.4}
  \pgfmathsetmacro{\sourceb}{3.6}
  \pgfmathsetmacro{\speeda}{speed(\sourcea)}
  \pgfmathsetmacro{\speedb}{speed(\sourceb)}

  % Panel (a): the same flow before the order Jacobian changes sign.
  \begin{scope}
    \node[anchor=west,font=\bfseries] at (0,6.35) {(a) Ordered};

    \draw[thick] (0,5) -- (12,3.2) node[above left] {surface $s(x)$};
    \draw[thick] (0,0.55) .. controls (2.2,0.48) and (4.0,0.62) .. (5.2,0.88)
      .. controls (6.0,1.08) and (7.0,0.62) .. (8.0,0.50)
      .. controls (9.2,0.38) and (11.0,0.44) .. (12,0.34) node[above left] {bed};

    \foreach \q in {0.7,1.5,2.2,4.2,5.1} {
      \pgfmathsetmacro{\surfacez}{5-0.15*\q}
      \draw[->] (\q,5.75) -- (\q,\surfacez+0.18);
    }
    \node[labelbox] at (5.45,5.55) {$\dot a>0$};

    \draw[isochrone]
      plot[domain=0:6,samples=100,smooth,variable=\q]
      ({\q+0.8*speed(\q)},{5-0.15*\q-0.24*speed(\q)});

    \coordinate (aorigin0) at (\sourcea,{5-0.15*\sourcea});
    \coordinate (aorigin1) at (\sourceb,{5-0.15*\sourceb});
    \coordinate (apoint0) at ({\sourcea+0.8*\speeda},{5-0.15*\sourcea-0.24*\speeda});
    \coordinate (apoint1) at ({\sourceb+0.8*\speedb},{5-0.15*\sourceb-0.24*\speedb});

    \draw[trajectory] (aorigin0) -- (apoint0);
    \draw[trajectory] (aorigin1) -- (apoint1);

    \fill (aorigin0) circle (1.7pt);
    \fill (aorigin1) circle (1.7pt);
    \node[labelbox,above left] at (aorigin0) {$x_0$};
    \node[labelbox,above right] at (aorigin1) {$x_0+\delta x_0$};

    \fill (apoint0) circle (1.7pt);
    \fill (apoint1) circle (1.7pt);
    \node[labelbox,below left] at (apoint0) {$x$};
    \node[labelbox,above right] at (apoint1) {$x+\delta x$};

    \node[labelbox] at (7.8,3.05) {isochrone $\tau$};
    \node[labelbox] at (5.5,2.35)
      {$\displaystyle \mathcal{J}=\left.\frac{\partial x}{\partial x_0}\right|_\tau>0$};

    \draw[->] (0.15,0.18) -- (1.35,0.18) node[right] {$x$};
    \draw[->] (0.15,0.18) -- (0.15,1.30) node[above] {$z$};
  \end{scope}

  % Panel (b): later in the same steady flow, neighboring origins reverse
  % their horizontal order along the isochrone.
  \begin{scope}[xshift=7.10cm]
    \node[anchor=west,font=\bfseries] at (0,6.35) {(b) Overturned};

    \draw[thick] (0,5) -- (12,3.2) node[above left] {surface $s(x)$};
    \draw[thick] (0,0.55) .. controls (2.2,0.48) and (4.0,0.62) .. (5.2,0.88)
      .. controls (6.0,1.08) and (7.0,0.62) .. (8.0,0.50)
      .. controls (9.2,0.38) and (11.0,0.44) .. (12,0.34) node[above left] {bed};

    \foreach \q in {0.7,1.5,2.2,4.2,5.1} {
      \pgfmathsetmacro{\surfacez}{5-0.15*\q}
      \draw[->] (\q,5.75) -- (\q,\surfacez+0.18);
    }
    \node[labelbox] at (5.45,5.55) {$\dot a>0$};

    \draw[isochrone]
      plot[domain=0:6,samples=120,smooth,variable=\q]
      ({\q+2.5*speed(\q)},{5-0.15*\q-0.75*speed(\q)});

    \coordinate (borigin0) at (\sourcea,{5-0.15*\sourcea});
    \coordinate (borigin1) at (\sourceb,{5-0.15*\sourceb});
    \coordinate (bpoint0) at ({\sourcea+2.5*\speeda},{5-0.15*\sourcea-0.75*\speeda});
    \coordinate (bpoint1) at ({\sourceb+2.5*\speedb},{5-0.15*\sourceb-0.75*\speedb});

    \draw[trajectory] (borigin0) -- (bpoint0);
    \draw[trajectory] (borigin1) -- (bpoint1);

    \fill (borigin0) circle (1.7pt);
    \fill (borigin1) circle (1.7pt);
    \node[labelbox,above left] at (borigin0) {$x_0$};
    \node[labelbox,above right] at (borigin1) {$x_0+\delta x_0$};

    \foreach \q in {2.3,2.6,3.0,3.4,3.7} {
      \pgfmathsetmacro{\uq}{speed(\q)}
      \draw[red,very thick] ({\q+2.5*\uq},{5-0.15*\q-0.75*\uq}) circle (2.3pt);
    }
    \fill (bpoint0) circle (1.7pt);
    \fill (bpoint1) circle (1.7pt);
    \draw[red,very thick] (bpoint0) circle (2.3pt);
    \draw[red,very thick] (bpoint1) circle (2.3pt);
    \node[labelbox,below right] at (bpoint0) {$x$};
    \node[labelbox,above left] at (bpoint1) {$x+\delta x$};

    \node[labelbox] at (8.2,1.45) {isochrone $\tau$};
    \node[labelbox,text=red!75!black] at (5.7,2.45)
      {$\displaystyle \mathcal{J}=\left.\frac{\partial x}{\partial x_0}\right|_\tau<0$};

    \draw[->] (0.15,0.18) -- (1.35,0.18) node[right] {$x$};
    \draw[->] (0.15,0.18) -- (0.15,1.30) node[above] {$z$};
  \end{scope}

\end{tikzpicture}

\end{document}
